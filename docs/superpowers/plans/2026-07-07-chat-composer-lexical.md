**Goal:** Replace the chat textarea composer with a Lexical composer that supports start-only `@Agent` and `/Skill` chips, typed prefix dropdowns, toolbar compatibility, and an always-visible one-model selector.

**Tech Stack:** Electron, React 19, Vite, TypeScript, Lexical, `@lexical/react`, Floating UI React, Tailwind CSS, Vitest, React Testing Library, Playwright Electron E2E.

import type { ComposerValue, SerializedComposerValue } from './types';

export function serializeComposerValue(value: ComposerValue): SerializedComposerValue {
  const body = value.text.trim();
  const skillPrefix = value.skill ? `/${value.skill.name}` : '';
  const text = [skillPrefix, body].filter(Boolean).join(' ');

  return {
    text,
    targetAgentId: value.agent?.id ?? null,
  };
}

export function hasSendableComposerValue(value: ComposerValue, attachmentCount: number): boolean {
  const serialized = serializeComposerValue(value);
  return serialized.text.length > 0 || attachmentCount > 0;
}
```

import type { PrefixQuery, PrefixTrigger } from './types';

const LOGICAL_START_QUERY_PATTERN = /^\s*([@/])([^\s@/]*)$/;

export function findLogicalStartPrefixQuery(textBeforeCaret: string): PrefixQuery | null {
  const normalized = textBeforeCaret.replace(/\u00a0/g, ' ');
  const match = LOGICAL_START_QUERY_PATTERN.exec(normalized);
  if (!match) return null;

  return {
    trigger: match[1] as PrefixTrigger,
    query: match[2] ?? '',
  };
}
```

import { $applyNodeReplacement, TextNode, type EditorConfig, type LexicalNode, type NodeKey, type SerializedTextNode } from 'lexical';
import { cn } from '@/lib/utils';
import type { ComposerChip } from './types';

export type SerializedComposerChipNode = SerializedTextNode & {
  type: 'composer-chip';
  version: 1;
  chip: ComposerChip;
};

function chipLabel(chip: ComposerChip): string {
  return chip.kind === 'agent' ? `@${chip.name}` : `/${chip.name}`;
}

function chipClassName(chip: ComposerChip): string {
  return cn(
    'inline-flex max-w-[14rem] items-center rounded-md px-1.5 py-0.5 align-baseline text-meta font-semibold leading-normal',
    'select-none whitespace-nowrap border transition-colors',
    chip.kind === 'agent'
      ? 'border-primary/20 bg-primary/5 text-primary dark:bg-primary/10'
      : 'border-skill-bg/20 bg-skill-bg/14 text-skill-fg dark:bg-skill-bg/18 dark:text-skill-fg-dark',
  );
}

export class ComposerChipNode extends TextNode {
  __chip: ComposerChip;

  static getType(): string {
    return 'composer-chip';
  }

  static clone(node: ComposerChipNode): ComposerChipNode {
    return new ComposerChipNode(node.__chip, node.__key);
  }

  static importJSON(serializedNode: SerializedComposerChipNode): ComposerChipNode {
    return $createComposerChipNode(serializedNode.chip)
      .setFormat(serializedNode.format)
      .setDetail(serializedNode.detail)
      .setStyle(serializedNode.style);
  }

  constructor(chip: ComposerChip, key?: NodeKey) {
    super(chipLabel(chip), key);
    this.__chip = chip;
  }

  getPayload(): ComposerChip {
    return this.getLatest().__chip;
  }

  createDOM(config: EditorConfig): HTMLElement {
    const dom = super.createDOM(config);
    const chip = this.getPayload();
    dom.className = chipClassName(chip);
    dom.dataset.testid = chip.kind === 'agent' ? 'chat-composer-agent-chip' : 'chat-composer-skill-token';
    dom.dataset.composerChipKind = chip.kind;
    dom.dataset.composerChipName = chip.name;
    if (chip.kind === 'agent') dom.dataset.composerAgentId = chip.id;
    if (chip.kind === 'skill') dom.dataset.composerSkillManifestPath = chip.manifestPath;
    return dom;
  }

  updateDOM(prevNode: this, dom: HTMLElement, config: EditorConfig): boolean {
    const didUpdate = super.updateDOM(prevNode, dom, config);
    const chip = this.getPayload();
    dom.className = chipClassName(chip);
    dom.dataset.testid = chip.kind === 'agent' ? 'chat-composer-agent-chip' : 'chat-composer-skill-token';
    dom.dataset.composerChipKind = chip.kind;
    dom.dataset.composerChipName = chip.name;
    if (chip.kind === 'agent') {
      dom.dataset.composerAgentId = chip.id;
      delete dom.dataset.composerSkillManifestPath;
    } else {
      dom.dataset.composerSkillManifestPath = chip.manifestPath;
      delete dom.dataset.composerAgentId;
    }
    return didUpdate;
  }

  exportJSON(): SerializedComposerChipNode {
    return {
      ...super.exportJSON(),
      type: 'composer-chip',
      version: 1,
      chip: this.getPayload(),
    };
  }

  canInsertTextBefore(): boolean {
    return false;
  }

  canInsertTextAfter(): boolean {
    return false;
  }

  isTextEntity(): true {
    return true;
  }
}

export function $createComposerChipNode(chip: ComposerChip): ComposerChipNode {
  return $applyNodeReplacement(new ComposerChipNode(chip).setMode('token'));
}

export function $isComposerChipNode(node: LexicalNode | null | undefined): node is ComposerChipNode {
  return node instanceof ComposerChipNode;
}
```

import { $createParagraphNode, $createTextNode, $getRoot, type LexicalNode, type TextNode } from 'lexical';
import { $createComposerChipNode, $isComposerChipNode } from './ComposerChipNode';
import type { ComposerAgentChip, ComposerChip, ComposerSkillChip, ComposerValue, PrefixQuery } from './types';

function walkTextNodes(node: LexicalNode, visit: (node: TextNode) => void): void {
  if ('getChildren' in node && typeof node.getChildren === 'function') {
    for (const child of node.getChildren()) walkTextNodes(child, visit);
    return;
  }
  if ('getTextContent' in node && typeof node.getTextContent === 'function') {
    visit(node as TextNode);
  }
}

export function $readComposerValue(): ComposerValue {
  const root = $getRoot();
  let agent: ComposerAgentChip | null = null;
  let skill: ComposerSkillChip | null = null;
  const textParts: string[] = [];

  walkTextNodes(root, (node) => {
    if ($isComposerChipNode(node)) {
      const chip = node.getPayload();
      if (chip.kind === 'agent') agent = chip;
      if (chip.kind === 'skill') skill = chip;
      return;
    }
    textParts.push(node.getTextContent());
  });

  return {
    text: textParts.join('').replace(/^\s+/, ''),
    agent,
    skill,
  };
}

export function $replaceComposerValue(value: ComposerValue): void {
  const root = $getRoot();
  root.clear();
  const paragraph = $createParagraphNode();

  const chips: ComposerChip[] = [];
  if (value.agent) chips.push(value.agent);
  if (value.skill) chips.push(value.skill);

  chips.forEach((chip, index) => {
    if (index > 0) paragraph.append($createTextNode(' '));
    paragraph.append($createComposerChipNode(chip));
  });

  const normalText = value.text.replace(/^\s+/, '');
  if (normalText) {
    if (chips.length > 0) paragraph.append($createTextNode(' '));
    paragraph.append($createTextNode(normalText));
  }

  root.append(paragraph);
  paragraph.selectEnd();
}

export function removeActivePrefixQuery(text: string, query: PrefixQuery | null): string {
  if (!query) return text;
  const prefixText = `${query.trigger}${query.query}`;
  if (text.startsWith(prefixText)) return text.slice(prefixText.length).replace(/^\s+/, '');
  return text;
}
```

import { useEffect } from 'react';
import { cn } from '@/lib/utils';
import type { PrefixOption } from './types';

interface PrefixDropdownProps<TItem> {
  open: boolean;
  anchor: HTMLElement | null;
  title: string;
  loading?: boolean;
  error?: string | null;
  emptyLabel: string;
  highlightedIndex: number;
  options: PrefixOption<TItem>[];
  getTestId: (option: PrefixOption<TItem>) => string;
  onSelect: (item: TItem) => void;
}

export function PrefixDropdown<TItem>({
  open,
  anchor,
  title,
  loading = false,
  error = null,
  emptyLabel,
  highlightedIndex,
  options,
  getTestId,
  onSelect,
}: PrefixDropdownProps<TItem>) {
  const { refs, floatingStyles } = useFloating({
    open,
    placement: 'top-start',
    middleware: [offset(8), flip(), shift({ padding: 8 })],
    whileElementsMounted: autoUpdate,
  });

  useEffect(() => {
    refs.setReference(anchor);
  }, [anchor, refs]);

  if (!open) return null;

  return (
    <div
      ref={refs.setFloating}
      style={floatingStyles}
      role="listbox"
      aria-label={title}
      data-testid="chat-composer-prefix-menu"
      className="z-50 w-80 overflow-hidden rounded-2xl border border-black/10 bg-surface-modal p-1.5 shadow-xl dark:border-white/10"
    >
      <div className="px-3 py-2 text-tiny font-medium text-muted-foreground/80">
        {title}
      </div>
      <div className="max-h-72 overflow-y-auto">
        {loading ? (
          <div className="px-3 py-4 text-xs text-muted-foreground">Loading...</div>
        ) : error ? (
          <div className="px-3 py-4 text-xs text-destructive">{error}</div>
        ) : options.length === 0 ? (
          <div className="px-3 py-4 text-xs text-muted-foreground">{emptyLabel}</div>
        ) : options.map((option, index) => (
          <button
            key={option.id}
            type="button"
            role="option"
            aria-selected={index === highlightedIndex}
            data-testid={getTestId(option)}

```tsx
import { forwardRef, useCallback, useEffect, useImperativeHandle, useMemo, useRef, useState } from 'react';
import { LexicalComposer } from '@lexical/react/LexicalComposer';
import { ContentEditable } from '@lexical/react/LexicalContentEditable';
import { LexicalErrorBoundary } from '@lexical/react/LexicalErrorBoundary';
import { HistoryPlugin } from '@lexical/react/LexicalHistoryPlugin';
import { OnChangePlugin } from '@lexical/react/LexicalOnChangePlugin';
import { PlainTextPlugin } from '@lexical/react/LexicalPlainTextPlugin';
import { useLexicalComposerContext } from '@lexical/react/LexicalComposerContext';
import { COMMAND_PRIORITY_HIGH, KEY_ARROW_DOWN_COMMAND, KEY_ARROW_UP_COMMAND, KEY_BACKSPACE_COMMAND, KEY_DELETE_COMMAND, KEY_ENTER_COMMAND, KEY_ESCAPE_COMMAND, type EditorState } from 'lexical';
import { mergeRegister } from '@lexical/utils';
import { cn } from '@/lib/utils';
import type { AgentSummary } from '@/types/agent';
import type { QuickAccessSkill } from '@/types/skill';
import { ComposerChipNode } from './ComposerChipNode';
import { findLogicalStartPrefixQuery } from './prefix-rules';
import { PrefixDropdown } from './PrefixDropdown';
import { $readComposerValue, $replaceComposerValue, removeActivePrefixQuery } from './lexical-state';
import type { ComposerAgentChip, ComposerSkillChip, ComposerValue, PrefixOption, PrefixQuery } from './types';

export interface ChatLexicalComposerHandle {
  clear: () => void;
  focus: () => void;
  insertAgent: (agent: ComposerAgentChip) => void;
  insertSkill: (skill: ComposerSkillChip) => void;
}

interface ChatLexicalComposerProps {
  disabled: boolean;
  placeholder: string;
  agents: AgentSummary[];
  skills: QuickAccessSkill[];
  skillsLoading: boolean;
  skillsError: string | null;
  labels: {
    agentTitle: string;
    skillTitle: string;
    agentEmpty: string;
    skillEmpty: string;
    skillLoading: string;
  };
  onChange: (value: ComposerValue) => void;
  onSubmit: () => void;
  onEscape: () => void;
  onRequestSkills: () => void;
  onPreviewSkill: (skill: { name: string; manifestPath: string }) => void;
}

function toAgentChip(agent: AgentSummary): ComposerAgentChip {
  return {
    kind: 'agent',
    id: agent.id,
    name: agent.name,
    modelDisplay: agent.modelDisplay,
  };
}

function toSkillChip(skill: QuickAccessSkill): ComposerSkillChip {
  return { ...skill, kind: 'skill' };
}

function EditorBridgePlugin({
  disabled,
  agents,
  skills,
  skillsLoading,
  skillsError,
  labels,
  onChange,
  onSubmit,
  onEscape,
  onRequestSkills,
  onPreviewSkill,
  bridgeRef,
}: ChatLexicalComposerProps & { bridgeRef: React.MutableRefObject<ChatLexicalComposerHandle | null> }) {
  const [editor] = useLexicalComposerContext();
  const [rootElement, setRootElement] = useState<HTMLElement | null>(null);
  const [prefixQuery, setPrefixQuery] = useState<PrefixQuery | null>(null);
  const [highlightedIndex, setHighlightedIndex] = useState(0);

  const agentOptions = useMemo<PrefixOption<AgentSummary>[]>(() => agents.map((agent) => ({
    id: agent.id,
    label: `@${agent.name}`,
    description: agent.modelDisplay,
    item: agent,
  })), [agents]);

  const skillOptions = useMemo<PrefixOption<QuickAccessSkill>[]>(() => skills.map((skill) => ({
    id: `${skill.source}:${skill.name}`,
    label: `/${skill.name}`,
    description: skill.description,
    sourceLabel: skill.sourceLabel,
    item: skill,
  })), [skills]);

  const activeOptions = useMemo(() => {
    if (!prefixQuery) return [];
    const query = prefixQuery.query.trim().toLowerCase();
    const sourceOptions = prefixQuery.trigger === '@' ? agentOptions : skillOptions;
    if (!query) return sourceOptions;
    return sourceOptions.filter((option) => `${option.label} ${option.description ?? ''} ${option.sourceLabel ?? ''}`.toLowerCase().includes(query));
  }, [agentOptions, prefixQuery, skillOptions]);

  const publishChange = useCallback((editorState: EditorState) => {
    editorState.read(() => {
      const value = $readComposerValue();
      onChange(value);
      const query = findLogicalStartPrefixQuery(value.text);
      setPrefixQuery(query);
      if (query?.trigger === '/') onRequestSkills();
    });
  }, [onChange, onRequestSkills]);

  const applyAgent = useCallback((agent: ComposerAgentChip) => {
    editor.update(() => {
      const current = $readComposerValue();
      $replaceComposerValue({
        agent,
        skill: current.skill,
        text: removeActivePrefixQuery(current.text, prefixQuery),
      });
    });
    setPrefixQuery(null);
  }, [editor, prefixQuery]);

  const applySkill = useCallback((skill: ComposerSkillChip) => {
    editor.update(() => {
      const current = $readComposerValue();
      $replaceComposerValue({
        agent: current.agent,
        skill,
        text: removeActivePrefixQuery(current.text, prefixQuery),
      });
    });
    setPrefixQuery(null);
  }, [editor, prefixQuery]);

  useEffect(() => {
    bridgeRef.current = {
      clear: () => editor.update(() => $replaceComposerValue({ text: '', agent: null, skill: null })),
      focus: () => editor.focus(),
      insertAgent: applyAgent,
      insertSkill: applySkill,
    };
    return () => {
      bridgeRef.current = null;
    };
  }, [applyAgent, applySkill, bridgeRef, editor]);

  useEffect(() => editor.registerRootListener((rootElement) => {
    setRootElement(rootElement);
  }), [editor]);

  useEffect(() => {
    if (!rootElement) return;
    const handleClick = (event: MouseEvent) => {
      const target = event.target;
      if (!(target instanceof HTMLElement)) return;
      const chipElement = target.closest('[data-composer-chip-kind="skill"]');
      if (!(chipElement instanceof HTMLElement)) return;
      const skillName = chipElement.dataset.composerChipName;
      const manifestPath = chipElement.dataset.composerSkillManifestPath;
      if (skillName && manifestPath) onPreviewSkill({ name: skillName, manifestPath });
    };
    rootElement.addEventListener('click', handleClick);
    return () => rootElement.removeEventListener('click', handleClick);
  }, [onPreviewSkill, rootElement]);

  useEffect(() => mergeRegister(
    editor.registerCommand(KEY_ESCAPE_COMMAND, () => {
      if (prefixQuery) {
        setPrefixQuery(null);
        return true;
      }
      onEscape();
      return false;
    }, COMMAND_PRIORITY_HIGH),
    editor.registerCommand(KEY_ENTER_COMMAND, (event) => {
      if (!event || event.shiftKey || event.isComposing) return false;
      if (prefixQuery && activeOptions[highlightedIndex]) {
        const selected = activeOptions[highlightedIndex].item;
        if (prefixQuery.trigger === '@') applyAgent(toAgentChip(selected as AgentSummary));
        if (prefixQuery.trigger === '/') applySkill(toSkillChip(selected as QuickAccessSkill));
        return true;
      }
      event.preventDefault();
      onSubmit();
      return true;
    }, COMMAND_PRIORITY_HIGH),
    editor.registerCommand(KEY_ARROW_UP_COMMAND, (event) => {
      if (!prefixQuery || activeOptions.length === 0) return false;
      event?.preventDefault();
      setHighlightedIndex((index) => (index <= 0 ? activeOptions.length - 1 : index - 1));
      return true;
    }, COMMAND_PRIORITY_HIGH),
    editor.registerCommand(KEY_ARROW_DOWN_COMMAND, (event) => {
      if (!prefixQuery || activeOptions.length === 0) return false;
      event?.preventDefault();
      setHighlightedIndex((index) => (index + 1) % activeOptions.length);
      return true;
    }, COMMAND_PRIORITY_HIGH),
    editor.registerCommand(KEY_BACKSPACE_COMMAND, () => false, COMMAND_PRIORITY_HIGH),
    editor.registerCommand(KEY_DELETE_COMMAND, () => false, COMMAND_PRIORITY_HIGH),
  ), [activeOptions, applyAgent, applySkill, editor, highlightedIndex, onEscape, onSubmit, prefixQuery]);

  useEffect(() => {
    setHighlightedIndex(0);
  }, [prefixQuery?.trigger, prefixQuery?.query]);

  const dropdownOpen = !!prefixQuery && !disabled;

  return (
    <>
      <OnChangePlugin onChange={publishChange} ignoreHistoryMergeTagChange />
      <PrefixDropdown
        open={dropdownOpen}
        anchor={rootElement}
        title={prefixQuery?.trigger === '@' ? labels.agentTitle : labels.skillTitle}
        loading={prefixQuery?.trigger === '/' && skillsLoading}
        loadingLabel={labels.skillLoading}
        error={prefixQuery?.trigger === '/' ? skillsError : null}
        emptyLabel={prefixQuery?.trigger === '@' ? labels.agentEmpty : labels.skillEmpty}
        highlightedIndex={highlightedIndex}
        options={activeOptions}
        getTestId={(option) => prefixQuery?.trigger === '@'
          ? `chat-composer-prefix-option-agent-${option.id}`
          : `chat-composer-prefix-option-skill-${option.id.replace(/[^a-zA-Z0-9_-]/g, '-')}`}
        onSelect={(item) => {
          if (prefixQuery?.trigger === '@') applyAgent(toAgentChip(item as AgentSummary));
          if (prefixQuery?.trigger === '/') applySkill(toSkillChip(item as QuickAccessSkill));
        }}
      />
    </>
  );
}

export const ChatLexicalComposer = forwardRef<ChatLexicalComposerHandle, ChatLexicalComposerProps>((props, ref) => {
  const bridgeRef = useRef<ChatLexicalComposerHandle | null>(null);

  useImperativeHandle(ref, () => ({
    clear: () => bridgeRef.current?.clear(),
    focus: () => bridgeRef.current?.focus(),
    insertAgent: (agent) => bridgeRef.current?.insertAgent(agent),
    insertSkill: (skill) => bridgeRef.current?.insertSkill(skill),
  }), []);

  const initialConfig = useMemo(() => ({
    namespace: 'ClawXChatComposer',
    nodes: [ComposerChipNode],
    onError(error: Error) {
      throw error;
    },
    theme: {
      paragraph: 'm-0',
    },
  }), []);

  return (
    <LexicalComposer initialConfig={initialConfig}>
      <PlainTextPlugin
        contentEditable={(
          <ContentEditable
            data-testid="chat-composer-input"
            className={cn(
              'relative min-h-[48px] max-h-[240px] overflow-y-auto whitespace-pre-wrap break-words bg-transparent p-0 text-sm leading-relaxed outline-none placeholder:text-muted-foreground/60',
              props.disabled && 'cursor-not-allowed opacity-60',
            )}
            aria-placeholder={props.placeholder}
          />
        )}
        placeholder={props.placeholder ? <div className="pointer-events-none absolute text-sm text-muted-foreground/60">{props.placeholder}</div> : null}
        ErrorBoundary={LexicalErrorBoundary}
      />
      <HistoryPlugin />
      <EditorBridgePlugin {...props} bridgeRef={bridgeRef} />
    </LexicalComposer>
  );
});

ChatLexicalComposer.displayName = 'ChatLexicalComposer';
```

function stableStringify(value: unknown): string {
  if (value == null || typeof value !== 'object') return JSON.stringify(value);
  if (Array.isArray(value)) return `[${value.map((item) => stableStringify(item)).join(',')}]`;
  const entries = Object.entries(value as Record<string, unknown>)

}
